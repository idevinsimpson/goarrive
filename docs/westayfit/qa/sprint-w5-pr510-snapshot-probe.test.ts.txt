type Goal = Record<string, unknown> & { goalId: string; sharedTotal: number | null; ownCredit: number | null };
type Community = Record<string, unknown> & { groupId: string; goals: Goal[] | null; partial: boolean };
type Snapshot = { schemaVersion: number; communities: Community[]; truncated: { communities: boolean; goals: boolean }; nextCursor: string | null };

const tag = () => randomBytes(5).toString('hex');
const DAY = 86_400_000;
const log = (label: string, v: unknown) => console.log(`W5PROBE ${label} ${JSON.stringify(v)}`);

async function seedGroup(name: string, extra: Record<string, unknown> = {}): Promise<string> {
  const ref = getFirestore().collection('wsfCommunityGroups').doc(`w5-${tag()}`);
  await ref.set({
    displayName: name, groupType: 'custom', joinPolicy: 'private', joinCode: `JCSECRET${tag()}`,
    lifecycleStatus: 'active', isSample: false, createdByUserId: 'creator-secret-uid', ...extra,
  });
  return ref.id;
}
async function seedMember(groupId: string, uid: string, membershipStatus = 'active', role = 'member') {
  await getFirestore().doc(`wsfMemberships/${groupId}_${uid}`).set({
    groupId, userId: uid, role, membershipStatus, nameVisibility: 'private', email: 'member-secret@example.com',
  });
}
async function seedGoal(groupId: string, opts: { status?: 'active' | 'closed'; total?: number; endsInDays?: number; id?: string; extra?: Record<string, unknown> } = {}) {
  const db = getFirestore();
  const ref = db.collection('wsfGoals').doc(opts.id ?? `g-${tag()}`);
  const now = Date.now();
  const doc: Record<string, unknown> = {
    ownerUid: 'champion-secret-uid', communityGroupId: groupId, title: `Goal ${ref.id}`, target: 500, unit: 'squats',
    status: opts.status ?? 'active', startsAt: Timestamp.fromMillis(now - 7 * DAY),
    endsAt: Timestamp.fromMillis(now + (opts.endsInDays ?? 7) * DAY), timezone: 'America/New_York',
    aggregateDisplayAuthorized: true, reachedAttemptId: null, ...(opts.extra ?? {}),
  };
  if (opts.status === 'closed') doc.closedAt = Timestamp.fromMillis(now - DAY);
  await ref.set(doc);
  if (opts.total) await db.doc(`wsfGoalCounters/${ref.id}/shards/3`).set({ count: opts.total });
  return ref.id;
}
async function seedOwnTs(goalId: string, uid: string, total: unknown, updatedAt: Timestamp | FieldValue) {
  await getFirestore().doc(`wsfGoalMemberTotals/${goalId}_${uid}`).set({ goalId, userId: uid, total, contributionCount: 1, updatedAt });
}
const seedOwn = (g: string, u: string, total: unknown, ms: number) => seedOwnTs(g, u, total, Timestamp.fromMillis(ms));

function call(uid: string | null, data: unknown = {}): Promise<Snapshot> {
  const request = { data, auth: uid ? { uid, token: {} } : undefined, rawRequest: {} } as never;
  return (wsfMyMemberSnapshot as unknown as { run: (r: never) => Promise<Snapshot> }).run(request);
}
async function outcome(p: Promise<unknown>): Promise<string> {
  try { await p; return 'RESOLVED'; } catch (e) {
    return e instanceof HttpsError ? `HttpsError:${e.code}` : `NonHttpsError:${(e as Error)?.constructor?.name}:${(e as Error)?.message}`;
  }
}
const goalsOf = (s: Snapshot) => s.communities.flatMap((c) => c.goals ?? []);
const byId = (s: Snapshot) => new Map(goalsOf(s).map((g) => [g.goalId, g]));

function keyPaths(v: unknown, path = ''): string[] {
  if (Array.isArray(v)) return v.flatMap((x) => keyPaths(x, `${path}[]`));
  if (v && typeof v === 'object') {
    return Object.entries(v as Record<string, unknown>).flatMap(([k, x]) => {
      const p = path ? `${path}.${k}` : k;
      return [p, ...keyPaths(x, p)];
    });
  }
  return [];
}

]);

// ── read instrumentation (in-process: .run executes in this Jest process) ──
const origGetAll = Firestore.prototype.getAll;
const origQueryGet = Query.prototype.get;
const origCount = Query.prototype.count;
function instrument() {
  const r = { rpcs: 0, docs: 0, byKind: {} as Record<string, { rpcs: number; docs: number }> };
  const bump = (k: string, docs: number) => {
    r.rpcs += 1; r.docs += docs;
    const e = (r.byKind[k] ??= { rpcs: 0, docs: 0 }); e.rpcs += 1; e.docs += docs;
  };
  const a = jest.spyOn(Firestore.prototype, 'getAll').mockImplementation(async function (this: Firestore, ...refs: Array<{ path: string }>) {
    const out = await origGetAll.apply(this, refs as never);
    const kinds = new Set(refs.map((x) => x.path.split('/')[0]));
    bump(`getAll:${[...kinds].join('+')}`, out.length);
    return out;
  } as never);
  const g = jest.spyOn(Query.prototype, 'get').mockImplementation(async function (this: Query) {
    const out = await origQueryGet.apply(this);
    bump(`query:${(this as unknown as { _queryOptions: { collectionId: string } })._queryOptions.collectionId}`, Math.max(out.size, 1));
    return out;
  } as never);
  const c = jest.spyOn(Query.prototype, 'count').mockImplementation(function (this: Query) {
    const coll = (this as unknown as { _queryOptions: { collectionId: string } })._queryOptions.collectionId;
    const agg = origCount.apply(this);
    const aggGet = agg.get.bind(agg);
    (agg as { get: unknown }).get = async () => { bump(`count:${coll}`, 1); return aggGet(); };
    return agg;
  } as never);
  return { r, restore: () => { a.mockRestore(); g.mockRestore(); c.mockRestore(); } };
}

  async function runP6(label: string, sameInstant: (k: number) => Timestamp | FieldValue, opts: { batch?: boolean } = {}) {
    const uid = `w5p6-${tag()}`;
    const g = await seedGroup(`P6 ${label}`);
    await seedMember(g, uid);
    const seeded: string[] = [];
    for (let i = 1; i <= 22; i += 1) {
      const id = await seedGoal(g, { status: 'closed', total: 1, endsInDays: -1 });
      await seedOwnTs(id, uid, i, new Timestamp(1890000100 - i, 0));
      seeded.push(id);
    }
    const tail: string[] = [];
    for (let k = 0; k < 8; k += 1) tail.push(await seedGoal(g, { status: 'closed', total: 1, endsInDays: -1 }));
    if (opts.batch) {
      const batch = getFirestore().batch();
      tail.forEach((id, k) => batch.set(getFirestore().doc(`wsfGoalMemberTotals/${id}_${uid}`), { goalId: id, userId: uid, total: 23 + k, contributionCount: 1, updatedAt: sameInstant(k) }));
      await batch.commit();
    } else {
      for (let k = 0; k < 8; k += 1) await seedOwnTs(tail[k]!, uid, 23 + k, sameInstant(k));
    }
    seeded.push(...tail);
    // Ground truth: the callable's own order, no limit.
    const truth = await getFirestore().collection('wsfGoalMemberTotals').where('userId', '==', uid)
      .orderBy('updatedAt', 'desc').orderBy(FieldPath.documentId(), 'desc').get();
    const truthOrder = truth.docs.map((d) => d.get('goalId') as string);
    const stored = truth.docs.map((d) => { const t = d.get('updatedAt') as Timestamp; return { goal: (d.get('goalId') as string), s: t.seconds, ns: t.nanoseconds }; });
    const p1 = await call(uid);
    const p2 = p1.nextCursor ? await call(uid, { cursor: p1.nextCursor }) : null;
    const a = goalsOf(p1).map((x) => x.goalId);
    const b = p2 ? goalsOf(p2).map((x) => x.goalId) : [];
    const got = new Set([...a, ...b]);
    const skipped = seeded.filter((x) => !got.has(x));
    const dup = a.filter((x) => b.includes(x));
    const res = {
      label, seeded: seeded.length, page1: a.length, page2: b.length, unionReturned: got.size,
      skippedCount: skipped.length, skippedTruthPositions: skipped.map((x) => truthOrder.indexOf(x) + 1).sort((x, y) => x - y),
      skippedTails: skipped.map((x) => tail.indexOf(x)),
      duplicated: dup.length, page2NextCursor: p2?.nextCursor ?? null,
      cursor: p1.nextCursor ? Buffer.from(p1.nextCursor, 'base64url').toString() : null,
      row25: stored[24], tailStoredPrecision: stored.slice(22),
      page1EqualsTruthPrefix: JSON.stringify(a) === JSON.stringify(truthOrder.slice(0, 25)),
      page2Order: b.map((x) => truthOrder.indexOf(x) + 1),
    };
    log(`P6.${label}`, res);
    return res;
  }
